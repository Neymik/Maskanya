```bash
#!/usr/bin/env bash
# Build olcrtc for ZOV (linux-amd64) and operator workstation (host platform).
# Can be run from anywhere — resolves repo root via git.

set -euo pipefail

# Ensure go-installed binaries (mage) are reachable even if ~/go/bin isn't on PATH globally.

ROOT="$(git rev-parse --show-toplevel)"
UPSTREAM="$ROOT/experiments/b-webrtc/upstream"

test -d "$UPSTREAM" || { echo "FATAL: $UPSTREAM missing — run Task 2 first." >&2; exit 1; }

mkdir -p "$OUTDIR"
cd "$UPSTREAM"

# Update deps once (cached in Go modules cache thereafter).
go mod download

# mage cross builds 9 platforms; we only keep linux-amd64 + host.
mage cross

HOST_OS=$(go env GOOS)
HOST_ARCH=$(go env GOARCH)

cp -v "build/olcrtc-linux-amd64"            "$OUTDIR/maskanya-olcrtc-linux-amd64"
cp -v "build/olcrtc-${HOST_OS}-${HOST_ARCH}" "$OUTDIR/maskanya-olcrtc-${HOST_OS}-${HOST_ARCH}"
chmod +x "$OUTDIR"/maskanya-olcrtc-*

echo "Built:"
ls -la "$OUTDIR/"
EOF
chmod +x experiments/b-webrtc/scripts/build.sh
```

```bash
ROOT="$(git rev-parse --show-toplevel)"
cd "$ROOT"
HOST_BIN="experiments/b-webrtc/build/maskanya-olcrtc-$(go env GOOS)-$(go env GOARCH)"
```

```bash
mkdir -p experiments/b-webrtc/systemd
cat > experiments/b-webrtc/systemd/maskanya-olcrtc-bridge.service <<'EOF'
[Unit]
Description=Maskanya Channel B bridge (olcrtc server, wbstream/datachannel)
Documentation=https://github.com/openlibrecommunity/olcrtc
After=network-online.target
Wants=network-online.target

[Service]
Type=simple
EnvironmentFile=/etc/maskanya/olcrtc.env
ExecStart=/usr/local/bin/maskanya-olcrtc \
  -mode srv \
  -carrier ${OLCRTC_CARRIER} \
  -transport ${OLCRTC_TRANSPORT} \
  -id ${OLCRTC_ROOM_ID} \
  -client-id ${OLCRTC_CLIENT_ID} \
  -key ${OLCRTC_KEY} \
  -link direct \
  -dns 1.1.1.1:53 \
  -data /var/lib/maskanya-olcrtc
WorkingDirectory=/var/lib/maskanya-olcrtc
# systemd 249 (Ubuntu 22.04) doesn't accept nested StateDirectory paths.
# Point -data at the StateDirectory root; olcrtc creates its own subdirs there.
StateDirectory=maskanya-olcrtc
Restart=on-failure
RestartSec=10s

# Hardening (conservative for v0 — MemoryDenyWriteExecute deliberately omitted because
# some Go GC paths allocate executable pages; re-evaluate after stable runtime.)
DynamicUser=yes
NoNewPrivileges=true
ProtectSystem=strict
ProtectHome=true
PrivateTmp=true
PrivateDevices=true
ProtectKernelTunables=true
ProtectKernelModules=true
ProtectControlGroups=true
RestrictAddressFamilies=AF_UNIX AF_INET AF_INET6 AF_NETLINK
RestrictNamespaces=true
RestrictRealtime=true
LockPersonality=true

[Install]
WantedBy=multi-user.target
EOF
```

## Task 9: 5-minute sustained throughput test

- [ ] **Step 1: Repeated transfers**

```bash
START=$(date +%s)
TOTAL_BYTES=0
ERRORS=0
while [[ $(($(date +%s) - START)) -lt 300 ]]; do
  BYTES=$(curl --socks5-hostname 127.0.0.1:1080 -m 30 -s -o /dev/null -w "%{size_download}" \
    https://speed.cloudflare.com/__down?bytes=1048576 2>/dev/null || echo 0)
  if [[ "$BYTES" -gt 0 ]]; then
    TOTAL_BYTES=$((TOTAL_BYTES + BYTES))
  else
    ERRORS=$((ERRORS + 1))
  fi
  sleep 2
done
ELAPSED=$(($(date +%s) - START))
echo "Elapsed: ${ELAPSED}s | Total: $((TOTAL_BYTES / 1024 / 1024)) MiB | Avg: $((TOTAL_BYTES * 8 / ELAPSED / 1000)) kbps | Errors: $ERRORS"
```

Expected: loop completes ~300s without the script crashing. Record exact MiB / kbps / error count for RESULTS — these are informational, not gating.

Verdict heuristic (operator judgment):
- **PASS**: no disconnect; ≥ 100 kbps sustained average; ≤ 5 errors. (Parent spec line 355 says 0.5–2 Mbps is the realistic WebRTC-SFU range, so anything in or above that band is healthy.)

- **FAIL**: disconnect mid-test, SOCKS5 listener dies, or `systemctl status maskanya-olcrtc-bridge` shows the server restarted during the run.
